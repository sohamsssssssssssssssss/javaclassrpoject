cmake_minimum_required(VERSION 3.20)
project(jade_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(jade_engine src/engine.cpp src/gemm.cpp src/checkpoint.cpp src/dataset.cpp src/training.cpp)
target_include_directories(jade_engine PUBLIC include)
target_compile_options(jade_engine PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  find_library(ACCELERATE_FRAMEWORK Accelerate REQUIRED)
  target_link_libraries(jade_engine PUBLIC ${ACCELERATE_FRAMEWORK})
  target_compile_definitions(jade_engine PUBLIC JADE_HAS_ACCELERATE=1 ACCELERATE_NEW_LAPACK)
endif()
enable_testing()
add_executable(jade_engine_tests tests/test_engine.cpp)
target_link_libraries(jade_engine_tests PRIVATE jade_engine)
target_compile_options(jade_engine_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME jade_engine_tests COMMAND jade_engine_tests ${CMAKE_CURRENT_SOURCE_DIR}/../docs/engine-v2/tiny-parity.txt)
add_executable(jade_engine_bench benchmarks/bench.cpp)
target_link_libraries(jade_engine_bench PRIVATE jade_engine)
target_compile_options(jade_engine_bench PRIVATE -Wall -Wextra -Wpedantic)
add_executable(jade_engine_20m_smoke benchmarks/smoke20m.cpp)
target_link_libraries(jade_engine_20m_smoke PRIVATE jade_engine)
target_compile_options(jade_engine_20m_smoke PRIVATE -Wall -Wextra -Wpedantic)
add_executable(jade_engine_50m_smoke benchmarks/smoke50m.cpp)
target_link_libraries(jade_engine_50m_smoke PRIVATE jade_engine)
target_compile_options(jade_engine_50m_smoke PRIVATE -Wall -Wextra -Wpedantic)
add_executable(jade_chat src/chat.cpp)
target_link_libraries(jade_chat PRIVATE jade_engine)
target_compile_options(jade_chat PRIVATE -Wall -Wextra -Wpedantic)
add_executable(jade_pretraining_tests tests/test_pretraining.cpp)
target_link_libraries(jade_pretraining_tests PRIVATE jade_engine)
target_compile_options(jade_pretraining_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME jade_pretraining_tests COMMAND jade_pretraining_tests
  ${CMAKE_CURRENT_SOURCE_DIR}/../data/tokenizer/jade-bpe-1024.v1
  ${CMAKE_CURRENT_SOURCE_DIR}/../docs/pretraining-v1/tokenizer-golden.tsv)
add_executable(jade_pretrain src/pretrain.cpp)
target_link_libraries(jade_pretrain PRIVATE jade_engine)
target_compile_options(jade_pretrain PRIVATE -Wall -Wextra -Wpedantic)
